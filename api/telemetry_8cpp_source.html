96static void process_nmea(const char* line) {
97 bool has_real_fix = false;
98 float lat = 0.0f, lon = 0.0f;
99 int sats = 0;
100
101 struct minmea_sentence_gga gga;
102 if (minmea_parse_gga(&gga, line)) {
103 // Il fix_quality > 0 è l'unico vero segnale di posizione valida
104 if (gga.fix_quality > 0) {
105 lat = minmea_tocoord(&gga.latitude);
106 lon = minmea_tocoord(&gga.longitude);
107 sats = gga.satellites_tracked;
108 has_real_fix = true;
109 }
110 }
111
112 // Se abbiamo un fix, aggiorniamo e logghiamo
113 if (has_real_fix) {
114 uint32_t save = spin_lock_blocking(telemetry_lock);
115 telemetry_data.latitude = lat;
116 telemetry_data.longitude = lon;
117 telemetry_data.gps_status = true;
118 telemetry_data.satellites = sats;
119 last_fix_timestamp = millis();
120 spin_unlock(telemetry_lock, save);
121
122 // LOGGING SOLO QUI: Con fix garantito e coordinate non nulle
123 check_and_log_gps(lat, lon);
124 }
125}

162void telemetry_update() {
163 // 1. LOGICA DI TIMEOUT: Se è passato troppo tempo dall'ultimo fix, azzera lo stato.
164 // Questo garantisce che la UI e i LED tornino allo stato "Searching" correttamente.
165 uint32_t save = spin_lock_blocking(telemetry_lock);
166 if (telemetry_data.gps_status && (millis() - last_fix_timestamp > 3000)) {
167 telemetry_data.gps_status = false;
168 telemetry_data.satellites = 0;
169 // Opzionale: potresti voler azzerare anche lat/lon qui se non vuoi mostrare l'ultima posizione nota
170 }
171 spin_unlock(telemetry_lock, save);
172
173 // 2. Aggiornamento lento (Temp/Uptime) - Eseguito solo ogni 500ms
174 static uint32_t last_slow_update = 0;
175 if (millis() - last_slow_update > 500) {
176 telemetry_data.temp_c = peripherals_get_temperature();
177 telemetry_data.uptime_s = millis() / 1000;
178 last_slow_update = millis();
179 }
180
181 // 3. Gestione errori (Controllo salute)
182 if (!telemetry_is_healthy()) {
183 if (!error_reported) {
184 sys_manager_report_error(ERR_CAT_SENSORS, ERR_SENS_GPS_NO_DATA, false);
185 error_reported = true;
186 }
187 }
188
189 // 4. Processing rate-limited: max 64 byte a ciclo per evitare saturazione del bus
190 int read_count = 0;
191 while (Serial2.available() && read_count < 64) {
192 char c = (char)Serial2.read();
193 read_count++;
194
195 if (c == '$') nmea_idx = 0;
196 if (nmea_idx < sizeof(nmea_buffer) - 1) nmea_buffer[nmea_idx++] = c;
197
198 if (c == '\n') {
199 nmea_buffer[nmea_idx] = '\0';
200 if (nmea_idx > 5) {
201 Serial.print("RAW: "); Serial.print(nmea_buffer); // DEBUG - opzionale
202 process_nmea(nmea_buffer);
203 }
204 nmea_idx = 0;
205 }
206 }
