68void peripherals_update_led_fx(LedAnimState state, float val) {
69 bool needs_show = true;
70 static LedAnimState last_state = (LedAnimState)-1;
71 if (last_state != state) {
72 fill_solid(leds, NUM_LEDS, CRGB::Black);
73 last_state = state;
74 }
75
76 switch (state) {
77 case LED_ANIM_BOOT: {
78 static uint8_t pos = 0;
79 fadeToBlackBy(leds, NUM_LEDS, 20);
80 leds[pos % NUM_LEDS] = CRGB(150, 150, 150);
81 pos++;
82 break;
83 }
84 case LED_ANIM_SEARCHING: {
85 if ((millis() / 500) % 2 == 0) fill_solid(leds, NUM_LEDS, CRGB::Red);
86 else fill_solid(leds, NUM_LEDS, CRGB::Black);
87 break;
88 }
89 case LED_ANIM_FIX_OK: {
90 float brightness_factor = constrain((float)val * 2.0f, 0.2f, 1.0f);
91 uint8_t brightness = (uint8_t)(brightness_factor * 255);
92 fill_solid(leds, NUM_LEDS, CRGB(0, brightness, 0));
93 break;
94 }
95 case LED_ANIM_SIGNAL_LOW: {
96 fill_solid(leds, NUM_LEDS, CRGB(150, 50, 0));
97 break;
98 }
99 default:
100 needs_show = false;
101 break;
102 }
103 if (needs_show) FastLED.show();
104}

130void peripherals_auto_feedback(const SystemDataPacket* data) {
131 if (!global_cfg.leds_enabled) {
132 fill_solid(leds, NUM_LEDS, CRGB::Black);
133 FastLED.show();
134 return;
135 }
136
137 if (!boot_finished) {
138 peripherals_update_led_fx(LED_ANIM_BOOT, 0.0f);
139 if (millis() - boot_start_time > 3000) boot_finished = true;
140 } else {
141 // MODIFICA: Invece di usare il confronto <, usiamo un confronto esplicito
142 // Assumendo che 0 sia NO_FIX e 1 sia FIX_2D/3D (come da log)
143 if (data->gps_status == 0) {
144 peripherals_update_led_fx(LED_ANIM_SEARCHING, 0.0f);
145 } else if (data->battery_v < 3.4f) {
146 peripherals_update_led_fx(LED_ANIM_SIGNAL_LOW, 0.0f);
147 } else {
148 // Qui entriamo se status >= 1
149 float sat_quality = constrain((float)data->satellites / 8.0f, 0.1f, 1.0f);
150 peripherals_update_led_fx(LED_ANIM_FIX_OK, sat_quality);
151 }
152 }
