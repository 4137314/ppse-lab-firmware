100void core0_setup() {
101 Serial.begin(115200);
102 uint32_t start_time = millis();
103 while (!Serial && (millis() - start_time < 1000)) delay(100);
104
105 // Messaggi di boot condizionali (visibili solo in Debug)
106 #if defined(DEBUG) && DEBUG == 1
107 Serial.println("\n--- PPSE FIRMWARE SYSTEM BOOTING ---");
108 #endif
109
110 // 1. Inizializzazione configurazione (Applica i default/flag di compilazione)
111 config_init();
112
113 // if (LittleFS.begin()) {
114 // Serial.println("\n[SYSTEM] Filesystem montato.");
115 // debug_dump_gps_log();
116 // } else {
117 // Serial.println("\n[SYSTEM] ERRORE: LittleFS non montato.");
118 // }
119delay(3000); // Aspetta 3 secondi, non 1
120
121 Serial.println("\n\n--- INIZIO TEST DI BOOT FORZATO ---");
122
123 // Prova ad inizializzare SOLO la Flash
124 if (LittleFS.begin()) {
125 Serial.println("LittleFS OK.");
126 } else {
127 Serial.println("LittleFS FALLITO.");
128 // Se fallisce qui, il file system è corrotto a livello hardware
129 }
130
131 // Ora prova a scrivere un file di test per vedere se la flash risponde
132 File f = LittleFS.open("/test.txt", "w");
133 if (f) {
134 f.println("Test scrittura riuscito");
135 f.close();
136 Serial.println("Scrittura test.txt OK.");
137 } else {
138 Serial.println("Scrittura test.txt FALLITA.");
139 }
140
141 // 2. Inizializzazione sottosistemi core
142 sys_manager_init();
143 telemetry_init();
144
145
146 // 3. Setup hardware di base
147 pinMode(LED_ALIVE_PIN, OUTPUT);
148 digitalWrite(LED_ALIVE_PIN, HIGH);
149
150 pinMode(BUCK_5V_EN_PIN, OUTPUT);
151 digitalWrite(BUCK_5V_EN_PIN, HIGH);
152
153 // 4. Inizializzazione periferiche (ora leggono correttamente global_cfg)
154 peripherals_init();
155 peripherals_init_leds();
156 inputs_init();
157
158 // 5. Setup UI
159 if (!display_hw_init()) {
160 sys_manager_report_error(ERR_CAT_HW, ERR_HW_DISPLAY_LOST, true);
161 } else {
162 run_hacker_boot_sequence(boot_led_update_wrapper);
163 }
164
165 ui_manager_init();
166
167 // Messaggio finale condizionale
168 #if defined(DEBUG) && DEBUG == 1
169 Serial.println("--- PPSE FIRMWARE SYSTEM READY ---");
170 #endif
171}
