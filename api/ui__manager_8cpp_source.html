50void ui_manager_update(const void* system_data) {
51 const SystemDataPacket* data = (const SystemDataPacket*)system_data;
52 Adafruit_SSD1306* canvas = (Adafruit_SSD1306*)get_display_driver();
53 if (!canvas) return;
54
55 if (data->last_error.code != last_error_code) {
56 error_acknowledged = false;
57 last_error_code = data->last_error.code;
58 // Feedback acustico istantaneo all'errore
59 #ifndef DEFAULT_BUZZER_OFF
60 if (last_error_code != 0) peripherals_trigger_feedback(FEEDBACK_ERROR);
61 #endif
62 }
63
64 canvas->clearDisplay();
65
66 if (data->flags.error_active && !error_acknowledged) {
67 render_error_overlay(data, canvas);
68 } else {
69 if (current_view && current_view->on_update) {
70 current_view->on_update(data);
71 }
72 }
73 canvas->display();
74}
