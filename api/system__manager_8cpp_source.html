40void sys_manager_report_error(error_category_t cat, error_code_t code, bool critical) {
41 uint32_t save = spin_lock_blocking(data_lock);
42
43 // Se l'errore è già attivo e identico, non fare nulla.
44 // Questo evita di forzare un aggiornamento continuo dell'UI.
45 if (shared_data.flags.error_active && shared_data.last_error.code == code) {
46 spin_unlock(data_lock, save);
47 return;
48 }
49
50 shared_data.last_error = {cat, code, millis(), critical};
51 shared_data.flags.error_active = 1;
52 spin_unlock(data_lock, save);
53}

69void sys_manager_handle_serial(void) {
70 if (Serial.available() > 0) {
71 String cmd = Serial.readStringUntil('\n');
72 cmd.trim();
73
74 // Comando per recuperare la posizione GPS
75 if (cmd == "GET_GPS") {
76 SystemDataPacket frame;
77 sys_manager_receive_data(&frame);
78
79 // LOGICA RESILIENTE: controlla se il fix è attivo, altrimenti usa la cache
80 if (frame.gps_status) {
81 // GPS LIVE: abbiamo segnale attuale
82 Serial.printf("GPS_DATA:LIVE,%.6f,%.6f\n", frame.latitude, frame.longitude);
83 } else {
84 // GPS NON DISPONIBILE: leggiamo dalla Flash
85 float lat, lon;
86 if (storage_read_last_gps(&lat, &lon)) {
87 Serial.printf("GPS_DATA:CACHE,%.6f,%.6f\n", lat, lon);
88 } else {
89 // Fallback totale in caso di errore di lettura
90 Serial.printf("GPS_DATA:CACHE,0.000000,0.000000\n");
91 }
92 }
93 }
94
95 // 2. Comando per ricezione dati meteo (WXC,City,Temp,Wind,Hum,Code)
96 else if (cmd.startsWith("WXC,")) {
97 int p1 = cmd.indexOf(',');
98 int p2 = cmd.indexOf(',', p1 + 1);
99 int p3 = cmd.indexOf(',', p2 + 1);
100 int p4 = cmd.indexOf(',', p3 + 1);
101 int p5 = cmd.indexOf(',', p4 + 1);
102
103 if (p5 != -1) {
104 WeatherDataPacket w;
105
106 cmd.substring(p1 + 1, p2).toCharArray(w.city, 32);
107 w.temp_ext = cmd.substring(p2 + 1, p3).toFloat();
108 w.wind_speed = cmd.substring(p3 + 1, p4).toFloat();
109 w.humidity = cmd.substring(p4 + 1, p5).toInt();
110 w.weather_code = cmd.substring(p5 + 1).toInt();
111 w.valid = true;
112
113 sys_manager_update_weather(w);
114
115 // === MODIFICA 2: PERSISTENZA IMMEDIATA IN LITTLEFS ===
116 storage_save_weather(&w);
117
118 Serial.println("OK_WXC");
119 }
120 }
121
122 // 3. Comando per stato diagnostico
123 else if (cmd == "GET_STATUS") {
124 SystemDataPacket frame;
125 sys_manager_receive_data(&frame);
126
127 Serial.println("--- STATO SISTEMA ---");
128 Serial.printf("Error Active: %d\n", frame.flags.error_active);
129
130 if(frame.flags.error_active) {
131 Serial.printf("Ultimo Errore: Cat 0x%02X, Code 0x%02X\n",
132 frame.last_error.category, frame.last_error.code);
133 } else {
134 Serial.println("System Healthy.");
135 }
136 }
137 }
138}
