26static void info_render(const void* data) {

28 Adafruit_SSD1306* canvas = (Adafruit_SSD1306*)get_display_driver();
29
30 canvas->clearDisplay();

32 canvas->setTextColor(SSD1306_WHITE);
33
34 // 0: Header
35 canvas->setCursor(0, 0);
36 canvas->print(F("INFO SISTEMA"));
37 canvas->drawFastHLine(0, 9, 128, SSD1306_WHITE);
38
39 // 1: FW Version (Inniettata a compilation time)
40 canvas->setCursor(0, 16);
41 canvas->print(F("VERSIONE: "));
42 canvas->print(FW_VERSION);
43
44 // 2: Uptime (Reattivo: il valore viene estratto dal pacchetto pkt)
45 canvas->setCursor(0, 32);
46 canvas->print(F("UPTIME: "));
47 canvas->printf("%02lu:%02lu:%02lu",
48 (pkt->uptime_s / 3600),
49 (pkt->uptime_s % 3600) / 60,
50 (pkt->uptime_s % 60));
51
52 // 3: Navigazione (Footer)
53 canvas->setCursor(0, 56);
