77void storage_log_gps_fix(float lat, float lon) {
78 // FILTRO DI SICUREZZA:
79 // Se le coordinate sono (0,0), ignoriamo la scrittura per non sporcare la cache.
80 // Usiamo una piccola tolleranza per sicurezza (es. < 0.0001)
81 if (lat > -0.0001 && lat < 0.0001 && lon > -0.0001 && lon < 0.0001) {
82 return;
83 }
84
85 File f = LittleFS.open("/gps.log", "a");
86 if (!f) {
87 Serial.println("[STORAGE] Errore apertura gps.log");
88 return;
89 }
90
91 f.printf("%lu,%.6f,%.6f\n", millis() / 1000, lat, lon);
92 f.close();
93
94 Serial.printf("[STORAGE] Posizione salvata: %.6f, %.6f\n", lat, lon);
95}

97bool storage_read_last_gps(float* lat, float* lon) {
98 File f = LittleFS.open("/gps.log", "r");
99 if (!f) {
100 Serial.println("[STORAGE DEBUG] ERRORE: Impossibile aprire /gps.log");
101 return false;
102 }
103
104 String last_line = "";
105 int line_count = 0;
106
107 Serial.println("[STORAGE DEBUG] Inizio scansione file...");
108
109 while (f.available()) {
110 String line = f.readStringUntil('\n');
111 if (line.length() > 5) {
112 last_line = line;
113 line_count++;
114 }
115 }
116 f.close();
117
118 Serial.printf("[STORAGE DEBUG] Scansione terminata. Righe trovate: %d\n", line_count);
119 Serial.printf("[STORAGE DEBUG] Ultima riga letta: '%s'\n", last_line.c_str());
120
121 if (last_line == "") {
122 Serial.println("[STORAGE DEBUG] ERRORE: File vuoto o corrotto");
123 return false;
124 }
125
126 int first_comma = last_line.indexOf(',');
127 int second_comma = last_line.indexOf(',', first_comma + 1);
128
129 if (first_comma == -1 || second_comma == -1) {
130 Serial.println("[STORAGE DEBUG] ERRORE: Formato riga non valido");
131 return false;
132 }
133
134 *lat = last_line.substring(first_comma + 1, second_comma).toFloat();
135 *lon = last_line.substring(second_comma + 1).toFloat();
136
137 Serial.printf("[STORAGE DEBUG] Parse riuscito: LAT %.6f, LON %.6f\n", *lat, *lon);
138 return true;
139}
