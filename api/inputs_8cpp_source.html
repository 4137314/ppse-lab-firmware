9// Rimosso IRAM_ATTR per compatibilità immediata
10void handle_button_interrupt() {
11 uint32_t current_time = millis();
12 if (current_time - last_interrupt_time > DEBOUNCE_DELAY_MS) {
13 if (digitalRead(BTN_UP_PIN) == LOW) pending_button = BTN_UP;
14 else if (digitalRead(BTN_DOWN_PIN) == LOW) pending_button = BTN_DOWN;
15 else if (digitalRead(BTN_OK_PIN) == LOW) pending_button = BTN_OK;
16 else if (digitalRead(BTN_BACK_PIN) == LOW) pending_button = BTN_BACK;
17
18 last_interrupt_time = current_time;
19 }
20}
