17static void settings_input(button_t btn, button_state_t state) {
18 if (state != BTN_RELEASED) return;
19
20 switch (btn) {
21 case BTN_UP:
22 cursor = (cursor == 0) ? SETTING_COUNT - 1 : cursor - 1;
23 break;
24 case BTN_DOWN:
25 cursor = (cursor + 1) % SETTING_COUNT;
26 break;
27 case BTN_BACK:
28 ui_manager_navigate_to(VIEW_ID_HOME);
29 break;
30 case BTN_OK:
31 #ifndef DEFAULT_BUZZER_OFF
32 peripherals_play_click();
33 #endif
34
35 switch(cursor) {
36 case SETTING_BRIGHTNESS:
37 global_cfg.oled_brightness = (global_cfg.oled_brightness >= 250) ? 50 : ((global_cfg.oled_brightness / 50) + 1) * 50;
38 display_set_brightness(global_cfg.oled_brightness);
39 break;
40 case SETTING_BUZZER:
41 global_cfg.buzzer_enabled = !global_cfg.buzzer_enabled;
42 peripherals_set_buzzer(global_cfg.buzzer_enabled);
43 break;
44 case SETTING_LEDS:
45 global_cfg.leds_enabled = !global_cfg.leds_enabled;
46 peripherals_set_leds(global_cfg.leds_enabled);
47 break;
48 case SETTING_ERASE:
49 LittleFS.end();
50 LittleFS.format();
51 LittleFS.begin();
52 erase_confirm = true;
53 confirm_timer = millis(); // Avvia il timer di 2 secondi
54 break;
55 case SETTING_INFO:
56 ui_manager_navigate_to(VIEW_ID_INFO);
57 break;
58 }
59 break;
60 default: break;
61 }
62}

91
92 if (cursor == i) {
93 canvas->setCursor(80, y);
94 if (i == SETTING_BRIGHTNESS) canvas->printf("%d", global_cfg.oled_brightness);
95 else if (i == SETTING_BUZZER) canvas->print(global_cfg.buzzer_enabled ? F("ON") : F("OFF"));
96 else if (i == SETTING_LEDS) canvas->print(global_cfg.leds_enabled ? F("ON") : F("OFF"));
97 else if (i == SETTING_ERASE && erase_confirm) canvas->print(F("OK"));
98 }
99 }
